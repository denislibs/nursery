Прогресс по тайлам

Каждый тайл уезжает в воркер через transfer без копирования, прогресс возвращается колбэком, результат переезжает обратно. Отмена доходит до цикла в воркере через AbortSignal.

const pool = createPool<typeof api>(() => new Worker(url, { type: 'module' }));

await map(tiles, (tile, _i, signal) => pool.api.blur(
  transfer({ pixels, w, h, radius }, [pixels.buffer]),
  { signal, onProgress: callback(p => bar(tile, p)) },
), { concurrency: pool.size, signal });
